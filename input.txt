# Coroutines: coro<TYield, TResume, TReturn>
# yield() is a method, not a keyword

coro<int, int> accumulator(int start) {
    int total = start;
    while (true) {
        total += yield(total);

}

coro<int, void, int> counter(int val) {
    int i = 0;
    while (i < val) {
        yield(i);
        i += 1;
    }
    return i;
}
